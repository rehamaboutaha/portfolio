Smart Parking & Safety System

An automated Arduino-powered parking system featuring real-time LCD slot monitoring, ultrasonic entrance sensors, IR exit detection, servo barrier controls, and emergency stop/open overrides.

Review Source Code 
#include <Wire.h>
#include <LiquidCrystal_I2C.h>
LiquidCrystal_I2C lcd(0x27, 16, 2);
#include <Servo.h>
Servo myservo;

int trig = 2;
int echo = 3;
int IR = 10;
int greenLed = 4;
int redLed = 5;
int buzzer = 6;
int emergencyStop = 8;
int emergencyOpen = 9;
int Slot = 4;

bool gateOpen = false;
bool emergencyMode = false;
bool emergencyOpenActive = false;
bool processingCar = false;
unsigned long gateActionTime = 0;
const unsigned long gateCloseDelay = 3000;
const unsigned long gateOpenDelay = 1000;
const int ENTRANCE_DISTANCE = 10;
int mode = 0;

long Ultrasonic() {
  digitalWrite(trig, LOW);
  delayMicroseconds(2);
  digitalWrite(trig, HIGH);
  delayMicroseconds(10);
  digitalWrite(trig, LOW);
  long duration = pulseIn(echo, HIGH);
  return duration * 0.034 / 2;
}

void setup() {
  Serial.begin(9600);
  lcd.init();
  lcd.backlight();
  pinMode(trig, OUTPUT);
  pinMode(echo, INPUT);
  pinMode(IR, INPUT_PULLUP);
  pinMode(greenLed, OUTPUT);
  pinMode(redLed, OUTPUT);
  pinMode(buzzer, OUTPUT);
  pinMode(emergencyStop, INPUT);
  pinMode(emergencyOpen, INPUT);
  myservo.attach(7);
  myservo.write(100);
  digitalWrite(greenLed, LOW);
  digitalWrite(redLed, LOW);
  noTone(buzzer);
  
  lcd.setCursor(0,0);
  lcd.print("  WELCOME TO   ");
  lcd.setCursor(0,1);
  lcd.print("REHAM PARKING ");
  delay(2500);
  lcd.clear();
}

void loop() {
  long distance = Ultrasonic();
  bool exitDetected = (digitalRead(IR) == LOW);
  bool stopPressed = (digitalRead(emergencyStop) == HIGH);
  bool openPressed = (digitalRead(emergencyOpen) == HIGH);

  if (stopPressed) {
    emergencyMode = true;
    emergencyOpenActive = false;
    processingCar = false;
    mode = 0;
    myservo.write(100);
    gateOpen = false;
    digitalWrite(redLed, HIGH);
    digitalWrite(greenLed, LOW);
    tone(buzzer, 1000);
    lcd.clear();
    lcd.setCursor(0,0);
    lcd.print(" EMERGENCY STOP ");
    lcd.setCursor(0,1);
    lcd.print("   ACTIVATED!   ");
    delay(500);
    return; 
  }

  if (openPressed) {
    emergencyMode = false;
    emergencyOpenActive = true;
    processingCar = false;
    mode = 0;
    myservo.write(0);
    gateOpen = true;
    digitalWrite(greenLed, HIGH);
    digitalWrite(redLed, LOW);
    noTone(buzzer);
    gateActionTime = millis();
    lcd.clear();
    lcd.setCursor(0,0);
    lcd.print(" EMERGENCY OPEN ");
    lcd.setCursor(0,1);
    lcd.print("   ACTIVATED!   ");
    delay(500);
    return; 
  }

  if (emergencyMode == true && !stopPressed) {
    emergencyMode = false;
    noTone(buzzer);
    digitalWrite(redLed, LOW);
  }

  if (!processingCar && !gateOpen && !emergencyMode && !emergencyOpenActive) {
    if (distance < ENTRANCE_DISTANCE && Slot > 0 && mode == 0) {
      mode = 1;
      processingCar = true;
      Slot--;
      if (Slot < 0) Slot = 0;
      myservo.write(0);
      gateOpen = true;
      gateActionTime = millis();
      digitalWrite(greenLed, HIGH);
      digitalWrite(redLed, LOW);
      noTone(buzzer);
      lcd.clear();
      lcd.setCursor(0,0);
      lcd.print(" CAR ENTERING   ");
      lcd.setCursor(0,1);
      lcd.print("Slots Left: ");
      lcd.print(Slot); 
    }
    else if (exitDetected && Slot < 4 && mode == 0) {
      mode = 2;
      processingCar = true;
      Slot++;
      if (Slot > 4) Slot = 4;
      myservo.write(0);
      gateOpen = true;
      gateActionTime = millis();
      digitalWrite(greenLed, HIGH);
      digitalWrite(redLed, LOW);
      noTone(buzzer);
      lcd.clear();
      lcd.setCursor(0,0);
      lcd.print(" CAR EXITING    ");
      lcd.setCursor(0,1);
      lcd.print("Slots Left: ");
      lcd.print(Slot); 
    }
    else if (distance < ENTRANCE_DISTANCE && Slot == 0) {
      lcd.clear();
      lcd.setCursor(0,0);
      lcd.print("   SORRY :(     ");
      lcd.setCursor(0,1);
      lcd.print(" PARKING FULL!  ");
      digitalWrite(redLed, HIGH);
      tone(buzzer, 800);
      delay(1500);
      noTone(buzzer);
      digitalWrite(redLed, LOW); 
    } 
  }

  if (gateOpen &&
      (millis() - gateActionTime > gateCloseDelay) &&
      !emergencyMode &&
      !emergencyOpenActive) 
  { 
    myservo.write(100);
    gateOpen = false;
    processingCar = false;
    mode = 0;
    digitalWrite(greenLed, LOW);
    lcd.clear();
    lcd.setCursor(0,0);
    lcd.print("WELCOME REHAM   ");
    lcd.setCursor(0,1);
    lcd.print("Slots Left: ");
    lcd.print(Slot);
  }

  if (!gateOpen && !processingCar && !emergencyMode && !emergencyOpenActive) {
    lcd.setCursor(0,0);
    lcd.print("WELCOME REHAM   ");
    lcd.setCursor(0,1);
    lcd.print("Slots Left: ");
    lcd.print(Slot);
    if (Slot == 0) {
      digitalWrite(redLed, HIGH);
    } else {
      digitalWrite(redLed, LOW);
    }
  } 
  delay(100);
}
